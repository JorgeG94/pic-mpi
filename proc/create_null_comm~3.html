   function create_null_comm() result(comm)
      type(comm_t) :: comm

      ! Explicitly initialize to null/invalid state
      comm%m_comm = MPI_COMM_NULL
      comm%m_rank = -1_int32
      comm%m_size = -1_int32
      comm%is_valid = .false.
   end function create_null_comm
